## 核心发现：问题 2 的根因

### 多型号并列宽表 0% 读取率的原因

**当前代码逻辑**（research_planning.py:1467-1468）：
```python
lines = [candidate_line.strip() for candidate_line in body.splitlines()
         if _is_candidate_family(candidate_line, query)][:200]
```

**只在"含候选型号的行"里匹配参数**，导致：

对于这种表格：
```
型号              H1650        H1650R       H2050R
最高转速(rpm)     16500        16500        20500
最大离心力(×g)    23950        23950        36900
```

1. 第 1 行（表头）含型号 `H1650R`，但没有数值
2. 第 2、3 行有数值，但不含型号 `H1650R`
3. 结果：参数值所在的行被完全跳过
4. **最终读取率：0%**

这正是用户反馈的"6~8 项里只读到 0~2 项"的**主要根因**！

```python
def _extract_from_multimodel_table(detail_lines, target_model, requirements):
    """从横向多型号并列宽表按列提取目标型号的参数值
    
    表格格式示例：
        型号              H1650        H1650R       H2050R
        最高转速(rpm)     16500        16500        20500
        最大离心力(×g)    23950        23950        36900
    
    返回：[(参数名, 值, 单位, 原文行), ...]
    """
    # 步骤 1：识别表头行（含多个型号）
    header_idx = None
    column_idx = None
    
    for i, line in enumerate(detail_lines):
        # 找出这一行里的所有型号
        models = extract_models_from_line(line)
        
        # 如果这一行含 2+ 个型号，且包含目标型号
        if len(models) >= 2 and target_model in models:
            header_idx = i
            # 估算目标型号所在的列位置（字符位置或分隔符计数）
            column_idx = estimate_column_position(line, target_model)
            break
    
    if header_idx is None or column_idx is None:
        return []
    
    # 步骤 2：后续行按参数名匹配，提取对应列的值
    results = []
    
    for i in range(header_idx + 1, min(len(detail_lines), header_idx + 30)):
        line = detail_lines[i]
        
        # 空行或分隔线跳过
        if not line.strip() or set(line.strip()) <= set('-=_| \t'):
            continue
        
        # 匹配要求中的参数名
        for req in requirements:
            labels = parameter_labels(req['name'])
            for label in labels:
                if label in line:
                    # 按列位置提取值
                    value_text = extract_column_value(line, column_idx)
                    
                    # 解析数值和单位
                    match = _MEASURED.search(value_text)
                    if match:
                        results.append({
                            'name': req['name'],
                            'value': as_number(match.group(1)),
                            'unit': canonical_unit(match.group(2)),
                            'evidence': line.strip(),
                        })
                    break
    
    return results


def estimate_column_position(line, target_model):
    """估算目标型号在表头行的列位置
    
    策略：
    1. 如果有明显分隔符（|、\t），按分隔符计数
    2. 否则按字符位置估算列宽
    """
    # 策略 1：按 | 或多个空格分隔
    if '|' in line or '\t' in line:
        parts = re.split(r'\||\t', line)
        for idx, part in enumerate(parts):
            if target_model in part:
                return idx
    
    # 策略 2：按字符位置估算
    model_pos = line.find(target_model)
    if model_pos >= 0:
        # 返回字符位置和估算列宽
        return {'char_pos': model_pos, 'width': 20}
    
    return None


def extract_column_value(line, column_idx):
    """从行中提取指定列的值
    
    支持：
    1. 按列序号提取（分隔符明确时）
    2. 按字符位置提取（固定列宽时）
    """
    # 情况 1：列序号
    if isinstance(column_idx, int):
        parts = re.split(r'\||\t', line)
        if 0 <= column_idx < len(parts):
            return parts[column_idx].strip()
    
    # 情况 2：字符位置
    if isinstance(column_idx, dict):
        char_pos = column_idx['char_pos']
        width = column_idx['width']
        return line[char_pos:char_pos + width].strip()
    
    return ''
```

```python
def extract_config_locally(text, requirements):
    """本地正则+表格解析提取配置，作为模型抽取的兜底
    
    策略：
    1. 正则匹配：参数名：数值 单位
    2. 表格解析：识别配置清单表格
    """
    results = []
    
    # 策略 1：正则匹配
    for req in requirements:
        labels = parameter_labels(req['name'])
        for label in labels:
            # 匹配：标签：数值 单位
            pattern = re.escape(label) + r'[：:]\s*([0-9.,]+)\s*([a-zA-Z℃×/]+)?'
            match = re.search(pattern, text)
            if match:
                results.append({
                    'name': req['name'],
                    'value': as_number(match.group(1)),
                    'unit': canonical_unit(match.group(2) or ''),
                    'source': 'regex',
                })
                break
    
    # 策略 2：表格解析（参考 price_table_rows 思路）
    # 识别"序号 | 参数名 | 技术要求"格式的表格
    table_results = parse_config_table(text, requirements)
    results.extend(table_results)
    
    return results


def parse_config_table(text, requirements):
    """解析配置清单表格
    
    表格示例：
        序号  技术参数名称      技术要求
        1     温度范围         -40℃~-86℃
        2     有效容积         578 L
    """
    results = []
    lines = text.splitlines()
    
    for i, line in enumerate(lines):
        for req in requirements:
            if req['name'] in line:
                # 提取同一行或下一行的值
                value_line = line
                if i + 1 < len(lines):
                    value_line += ' ' + lines[i + 1]
                
                match = _MEASURED.search(value_line)
                if match:
                    results.append({
                        'name': req['name'],
                        'value': as_number(match.group(1)),
                        'unit': canonical_unit(match.group(2)),
                        'source': 'table',
                    })
    
    return results
```
